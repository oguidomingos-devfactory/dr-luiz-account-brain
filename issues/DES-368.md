```typescript
import { useEffect, useRef } from 'react'
import gsap from 'gsap'
import { ScrollTrigger } from 'gsap/ScrollTrigger'

gsap.registerPlugin(ScrollTrigger)

export function HeroSection() {
  const containerRef = useRef<HTMLDivElement>(null)
  const titleRef = useRef<HTMLHeadingElement>(null)
  const subtitleRef = useRef<HTMLParagraphElement>(null)
  const imageRef = useRef<HTMLDivElement>(null)

  useEffect(() => {
    const timeline = gsap.timeline({
      scrollTrigger: {
        trigger: containerRef.current,
        start: 'top center',
        markers: false,
      },
    })

    // Fade in + scale heading
    timeline.from(titleRef.current, {
      opacity: 0,
      y: 30,
      scale: 0.95,
      duration: 0.8,
    }, 0)

    // Fade in subtitle with delay
    timeline.from(subtitleRef.current, {
      opacity: 0,
      y: 20,
      duration: 0.6,
    }, 0.2)

    // Image slide + fade
    timeline.from(imageRef.current, {
      opacity: 0,
      x: 40,
      duration: 0.8,
    }, 0.3)

    return () => {
      ScrollTrigger.getAll().forEach(trigger => trigger.kill())
    }
  }, [])

  return (
    <section ref={containerRef} className="py-20 md:py-32 bg-gradient-to-br from-medical-50 via-white to-blue-50">
      <div className="max-w-6xl mx-auto px-4 sm:px-6 lg:px-8">
        <div className="grid md:grid-cols-2 gap-12 items-center">
          {/* Left: Text */}
          <div>
            <h2
              ref={titleRef}
              className="text-4xl md:text-5xl font-serif font-bold text-medical-900 mb-6 leading-tight"
            >
              Especialista em Quadril & Dor
            </h2>
            <p
              ref={subtitleRef}
              className="text-lg text-gray-600 mb-8 leading-relaxed"
            >
              Com mais de 10 anos de experiência, trato pacientes com dor crônica, 
              lesões de quadril e condições ortopédicas complexas usando técnicas avançadas.
            </p>
            <button className="bg-accent-teal text-white px-8 py-3 rounded-lg font-medium hover:opacity-90 transition-opacity">
              Agende Agora
            </button>
          </div>

          {/* Right: Image placeholder */}
          <div
            ref={imageRef}
            className="bg-gradient-to-br from-medical-100 to-accent-teal/20 rounded-2xl h-96 flex items-center justify-center"
          >
            <span className="text-gray-400">Imagem do médico</span>
          </div>
```

```typescript
export interface LeadFormData {
  name: string
  email: string
  phone: string
  specialty: string
  message: string
  agreeTerms: boolean
}

export function validateForm(data: LeadFormData): Record<string, string> {
  const errors: Record<string, string> = {}

  if (!data.name || data.name.trim().length < 2) {
    errors.name = 'Nome é obrigatório (mín. 2 caracteres)'
  }

  if (!data.email || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(data.email)) {
    errors.email = 'Email inválido'
  }

  if (!data.phone || data.phone.replace(/\D/g, '').length < 10) {
    errors.phone = 'Telefone inválido (mín. 10 dígitos)'
  }

  if (!data.specialty) {
    errors.specialty = 'Selecione uma especialidade'
  }

  if (!data.agreeTerms) {
    errors.agreeTerms = 'Você deve aceitar os termos'
  }

  return errors
}
```

```typescript
interface GTMEvent {
  event: string
  [key: string]: any
}

export function trackEvent(eventData: GTMEvent): void {
  if (window.dataLayer) {
    window.dataLayer.push(eventData)
  }
}

export function trackFormSubmit(formName: string, specialty: string): void {
  trackEvent({
    event: 'form_submit',
    form_name: formName,
    specialty: specialty,
    timestamp: new Date().toISOString(),
  })
}

export function trackPageView(pageName: string): void {
  trackEvent({
    event: 'page_view',
    page_title: pageName,
    page_location: window.location.href,
  })
}

export function trackWhatsAppClick(): void {
  trackEvent({
    event: 'whatsapp_click',

}

  trackEvent({
    event: 'specialty_view',
    specialty: specialty,
```
